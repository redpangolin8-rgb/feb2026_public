Closing Price¶

In [3]:
fig, ax = plt.subplots()

for sym, df in data.items():
    ax.plot(df.index, df['Close'], label=sym, linewidth=1.5)

ax.set_title(f'Closing Price — last {PERIOD}')
ax.set_xlabel('Date')
ax.set_ylabel('Price (USD)')
ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %Y'))
ax.xaxis.set_major_locator(mdates.MonthLocator())
fig.autofmt_xdate()
ax.legend()
plt.tight_layout()
plt.show()

Normalised Performance (base 100)¶

fig, ax = plt.subplots()

for sym, df in data.items():
    close = df['Close']
    normalised = close / close.iloc[0] * 100
    ax.plot(df.index, normalised, label=sym, linewidth=1.5)

ax.axhline(100, color='gray', linestyle='--', linewidth=0.8)
ax.set_title(f'Normalised Performance (base 100) — last {PERIOD}')
ax.set_xlabel('Date')
ax.set_ylabel('Indexed Price')
ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %Y'))
ax.xaxis.set_major_locator(mdates.MonthLocator())
fig.autofmt_xdate()
ax.legend()
plt.tight_layout()
plt.show()

Candlestick Chart (AAPL)¶

In [5]:
sym = 'AAPL'
df  = data[sym].tail(60).copy()

up   = df[df['Close'] >= df['Open']]
down = df[df['Close'] <  df['Open']]

fig, ax = plt.subplots()

width  = 0.6   # candle body width in days
width2 = 0.05  # wick width

ax.bar(up.index,   up['Close']   - up['Open'],   width, bottom=up['Open'],   color='#26a69a')
ax.bar(up.index,   up['High']    - up['Close'],  width2, bottom=up['Close'],  color='#26a69a')
ax.bar(up.index,   up['Open']    - up['Low'],    width2, bottom=up['Low'],    color='#26a69a')

ax.bar(down.index, down['Open']  - down['Close'], width, bottom=down['Close'], color='#ef5350')
ax.bar(down.index, down['High']  - down['Open'],  width2, bottom=down['Open'], color='#ef5350')
ax.bar(down.index, down['Close'] - down['Low'],   width2, bottom=down['Low'],  color='#ef5350')

ax.set_title(f'{sym} Candlestick — last 60 sessions')
ax.set_xlabel('Date')
ax.set_ylabel('Price (USD)')
ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %d'))
ax.xaxis.set_major_locator(mdates.WeekdayLocator(byweekday=0))
fig.autofmt_xdate()
plt.tight_layout()
plt.show()

fig, axes = plt.subplots(len(SYMBOLS), 1, sharex=True, figsize=(12, 3 * len(SYMBOLS)))

for ax, sym in zip(axes, SYMBOLS):
    df = data[sym]
    colors = ['#26a69a' if c >= o else '#ef5350'
              for c, o in zip(df['Close'], df['Open'])]
    ax.bar(df.index, df['Volume'] / 1e6, color=colors, width=0.8)
    ax.set_ylabel(f'{sym}\nVolume (M)')
    ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %Y'))
    ax.xaxis.set_major_locator(mdates.MonthLocator())

axes[-1].set_xlabel('Date')
fig.suptitle(f'Daily Volume — last {PERIOD}', y=1.01)
fig.autofmt_xdate()
plt.tight_layout()
plt.show()
