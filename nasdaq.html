Closing Price¶

fig, ax = plt.subplots()

ax.plot(df.index, df['Close'], color='#1565c0', linewidth=1.5)
ax.fill_between(df.index, df['Close'], df['Close'].min(), alpha=0.1, color='#1565c0')

ax.set_title(f'NASDAQ Composite — Closing Price (last {PERIOD})')
ax.set_xlabel('Date')
ax.set_ylabel('Index Value')
ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %Y'))
ax.xaxis.set_major_locator(mdates.MonthLocator())
fig.autofmt_xdate()
plt.tight_layout()
plt.show()

Candlestick Chart¶

In [4]:
plot_df = df.tail(60).copy()

up   = plot_df[plot_df['Close'] >= plot_df['Open']]
down = plot_df[plot_df['Close'] <  plot_df['Open']]

fig, ax = plt.subplots()

width  = 0.6
width2 = 0.05

ax.bar(up.index,   up['Close']   - up['Open'],    width,  bottom=up['Open'],    color='#26a69a')
ax.bar(up.index,   up['High']    - up['Close'],   width2, bottom=up['Close'],   color='#26a69a')
ax.bar(up.index,   up['Open']    - up['Low'],     width2, bottom=up['Low'],     color='#26a69a')

ax.bar(down.index, down['Open']  - down['Close'], width,  bottom=down['Close'], color='#ef5350')
ax.bar(down.index, down['High']  - down['Open'],  width2, bottom=down['Open'],  color='#ef5350')
ax.bar(down.index, down['Close'] - down['Low'],   width2, bottom=down['Low'],   color='#ef5350')

ax.set_title('NASDAQ Composite — Candlestick (last 60 sessions)')
ax.set_xlabel('Date')
ax.set_ylabel('Index Value')
ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %d'))
ax.xaxis.set_major_locator(mdates.WeekdayLocator(byweekday=0))
fig.autofmt_xdate()
plt.tight_layout()
plt.show()

Daily Returns Distribution¶

In [5]:
returns = df['Close'].pct_change().dropna() * 100  # percent

fig, ax = plt.subplots()

colors = ['#26a69a' if r >= 0 else '#ef5350' for r in returns]
ax.bar(returns.index, returns, color=colors, width=0.8)
ax.axhline(0, color='gray', linewidth=0.8, linestyle='--')

ax.set_title(f'NASDAQ Composite — Daily Returns % (last {PERIOD})')
ax.set_xlabel('Date')
ax.set_ylabel('Return (%)')
ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %Y'))
ax.xaxis.set_major_locator(mdates.MonthLocator())
fig.autofmt_xdate()
plt.tight_layout()
plt.show()
